Integration Guide

To send automated notifications from your billing, webhook, or website apps, make a POST request to the endpoint below. Messages are processed sequentially through a queue to prevent spam blocks.

API Endpoint URL

http://domain.com/api/send

Method

POST

Headers

Content-Type: application/json

Payload JSON Example

{
  "number": "6287815873285",
  "message": "Halo! Tagihan invoice #10294 Anda telah lunas.",
  "device_id": "1" // Optional: "1" to "10", defaults to auto load-balancing
}

Integration Code Snippets

<?php
$url = "http://domain.com/api/send";
$data = [
    "number" => "6287815873285",
    "message" => "Halo! Tagihan invoice #10294 Anda telah lunas.",
    "device_id" => "1" // opsional
];

$ch = curl_init($url);
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
$response = curl_exec($ch);
curl_close($ch);
echo $response;
?>

import requests

url = "http://domain.com/api/send"
payload = {
    "number": "6287815873285",
    "message": "Halo! Tagihan invoice #10294 Anda telah lunas.",
    "device_id": "1"
}
response = requests.post(url, json=payload)
print(response.json())
const fetch = require('node-fetch');

const url = "http://domain.com/api/send";
const payload = {
  number: "6287815873285",
  message: "Halo! Tagihan invoice #10294 Anda telah lunas.",
  device_id: "1"
};

fetch(url, {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify(payload)
})
.then(res => res.json())
.then(json => console.log(json))
.catch(err => console.error(err));
